        window.saveSet = function(day, exIdx, setNum) {

            const storageKey = `day_${day}_ex_${exIdx}_set_${setNum}`;

            const weight = document.getElementById(`w_${day}_${exIdx}_${setNum}`).value;

            const reps = document.getElementById(`r_${day}_${exIdx}_${setNum}`).value;

        window.toggleSet = function(day, exIdx, setNum) {

            const btn = document.getElementById(`btn_${day}_${exIdx}_${setNum}`);

            btn.innerText = btn.classList.contains('logged') ? '✓' : 'Log';

            saveSet(day, exIdx, setNum);

        };
